    46→function formatSize(bytes: number): string {
    47→  const mb = bytes / (1024 * 1024)
    48→  if (mb >= 1) return `${mb.toFixed(1)} MB`
    49→  return `${Math.max(1, Math.round(bytes / 1024))} KB`
    50→}

   182→      setPdfBoard('')

   192→  function addChapter() {
   193→    setFormError(null)
   194→    setSuccessMsg(null)
   195→    const t = chTitle.trim()

   197→    if (!t) {
   198→      setFormError('অধ্যায়ের শিরোনাম দাও।')
   199→      return

   201→    if (c.length < 50) {
   202→      setFormError('লেখা খুব ছোট — অন্তত ৫০ অক্ষরের লেখা পেস্ট করো।')
   203→      return
   204→    }
   205→    const num = chNumber.trim() === '' ? undefined : Number(chNumber)
   206→    const page = chPageStart.trim() === '' ? undefined : Number(chPageStart)
   207→    if (num !== undefined && (!Number.isFinite(num) || num <= 0)) {
   208→      setFormError('অধ্যায় নম্বর সঠিক পজিটিভ সংখ্যা দাও।')
   209→      return
   210→    }
   211→    if (page !== undefined && (!Number.isFinite(page) || page <= 0)) {
   212→      setFormError('শুরু পৃষ্ঠা সঠিক পজিটিভ সংখ্যা দাও।')
   213→      return
   214→    }
   215→    setChapters((prev) => [...prev, { title: t, number: num, pageStart: page, content: c }])
   216→    setChTitle('')
   217→    setChNumber('')
   218→    setChPageStart('')
   219→    setChContent('')
   220→  }

   274→  async function handleEmbed(book: BookInfo) {
   275→    if (busyBookId || book.autoEmbedding) return
   276→    setEmbedError(null)
   277→    setEmbedResult(null)

   280→      const data = await api<{ embeddedCount: number; remainingCount: number }>('/api/admin/embed', {
   281→        method: 'POST',
   282→        body: { bookId: book.id },
   283→      })
   284→      setEmbedResult({
   285→        bookId: book.id,
   286→        embeddedCount: data.embeddedCount,
   287→        remainingCount: data.remainingCount,
   288→      })
   289→      await loadBooks()
   290→    } catch (e) {
   291→      setEmbedError(e instanceof ApiError ? e.message : 'এমবেডিং করা গেল না, আবার চেষ্টা করো।')
   292→    } finally {
   293→      setBusyBookId(null)

   309→  async function doDelete(id: string) {

   317→    }
